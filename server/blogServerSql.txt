--创建用户信息表
CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY comment '用户唯一ID（主键）',      
  username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名（唯一）',    
  password_hash VARCHAR(255) NOT NULL COMMENT '加密密码', 
  email VARCHAR(100) UNIQUE  COMMENT '邮箱（唯一）',
  phone VARCHAR(20) COMMENT '手机号（可选）',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '注册时间',  
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '最后更新时间',  
  status TINYINT DEFAULT 1 COMMENT '账户状态:0:禁用,1:启用'  
) COMMENT='用户信息表';

-- 创建留言表 (messageBoard页面用)
CREATE TABLE messages (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT COMMENT '用户ID（可为空，表示用户已删除）', -- 注意去掉NOT NULL
    message_content TEXT NOT NULL COMMENT '留言内容',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE /*级联删除，user表中用户删除后，message表中id对应的相关数据一起删除*/
);

-- 创建回复表
CREATE TABLE replies (
  id INT AUTO_INCREMENT PRIMARY KEY COMMENT '回复唯一ID',
  message_id INT NOT NULL COMMENT '关联留言ID',
  user_id INT NOT NULL COMMENT '回复用户ID',
  content TEXT NOT NULL COMMENT '回复内容',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '回复时间',
  FOREIGN KEY (message_id) REFERENCES messages(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  INDEX idx_message_created (message_id, created_at)
) COMMENT='回复表';


--收藏夹
-- 创建主分类表（存储分类标题）
CREATE TABLE favorite_categories (
  id INT PRIMARY KEY AUTO_INCREMENT COMMENT '主分类 ID',
  title VARCHAR(50) NOT NULL UNIQUE COMMENT '分类标题（如 "常用"）',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
);
-- 创建收藏夹子项表（存储每个分类下的具体条目）
CREATE TABLE favorite_items (
  id INT PRIMARY KEY AUTO_INCREMENT COMMENT '子项 ID',
  category_id INT NOT NULL COMMENT '关联的主分类 ID',
  imgSrc VARCHAR(255) NOT NULL COMMENT '图片地址',
  subtitle VARCHAR(50) NOT NULL COMMENT '子项副标题（如 "CSS可视化"）',
  introduce TEXT NOT NULL COMMENT '子项描述',
  href VARCHAR(255) NOT NULL COMMENT '跳转链接',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  FOREIGN KEY (category_id) REFERENCES favorite_categories(id) ON DELETE CASCADE
);
