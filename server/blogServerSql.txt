--创建用户信息表(登录/注册)
CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY comment '用户唯一ID（主键）',      
  username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名（唯一）',    
  password_hash VARCHAR(255) NOT NULL COMMENT '加密密码', 
  email VARCHAR(100) UNIQUE  COMMENT '邮箱（唯一）',
  phone VARCHAR(20) COMMENT '手机号（可选）',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '注册时间',  
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '最后更新时间',  
  status TINYINT DEFAULT 1 COMMENT '账户状态:0:禁用,1:启用'  
) COMMENT='用户信息表';

-- 创建留言板数据表 (messageBoard页面用)
CREATE TABLE messages (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT COMMENT '用户ID（可为空，表示用户已删除）', -- 注意去掉NOT NULL
    message_content TEXT NOT NULL COMMENT '留言内容',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE /*级联删除，user表中用户删除后，message表中id对应的相关数据一起删除*/
);


-- 创建回复表
CREATE TABLE replies (
  id INT AUTO_INCREMENT PRIMARY KEY COMMENT '回复唯一ID',
  message_id INT NOT NULL COMMENT '关联留言ID',
  user_id INT NOT NULL COMMENT '回复用户ID',
  content TEXT NOT NULL COMMENT '回复内容',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '回复时间',
  FOREIGN KEY (message_id) REFERENCES messages(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  INDEX idx_message_created (message_id, created_at)
) COMMENT='回复表';


--收藏夹
-- 创建主分类表（存储分类标题）
CREATE TABLE favorite_categories (
  id INT PRIMARY KEY AUTO_INCREMENT COMMENT '主分类 ID',
  title VARCHAR(50) NOT NULL UNIQUE COMMENT '分类标题（如 "常用"）',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
);
-- 创建收藏夹子项表（存储每个分类下的具体条目）
CREATE TABLE favorite_items (
  id INT PRIMARY KEY AUTO_INCREMENT COMMENT '子项 ID',
  category_id INT NOT NULL COMMENT '关联的主分类 ID',
  imgSrc VARCHAR(255) NOT NULL COMMENT '图片地址',
  subtitle VARCHAR(50) NOT NULL COMMENT '子项副标题（如 "CSS可视化"）',
  introduce TEXT NOT NULL COMMENT '子项描述',
  href VARCHAR(255) NOT NULL COMMENT '跳转链接',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  FOREIGN KEY (category_id) REFERENCES favorite_categories(id) ON DELETE CASCADE
);

--博客内容分类表
CREATE TABLE blog_content_class (
  id BIGINT UNSIGNED AUTO_INCREMENT COMMENT '主键ID（自增）',
  order_id VARCHAR(36) NOT NULL COMMENT '关联查询ID（如UUID，36位）',
  title VARCHAR(50) NOT NULL COMMENT '导航标题（最长50字符）',
  subtitle VARCHAR(50) NOT NULL COMMENT '内容简概（最长50字符）',
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  update_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (id),
  INDEX idx_create_time (create_time) COMMENT '创建时间索引（用于按时间范围查询）'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='博客内容分类表';
--博客内容主体
CREATE TABLE blog_content_main (
  id BIGINT UNSIGNED AUTO_INCREMENT COMMENT '主键ID（自增，全局唯一标识）',
  father_id BIGINT UNSIGNED COMMENT '父内容ID（关联自身表id，NULL表示无父级）', -- 移除 NOT NULL
  biz_id VARCHAR(36) NOT NULL UNIQUE COMMENT '业务自有ID（如UUID，全局唯一）',
  author_id VARCHAR(50) NOT NULL COMMENT '关联作者ID（最长50字符）',
  subtitle VARCHAR(500) NOT NULL COMMENT '内容简概（最长500字符）',
  title VARCHAR(200) NOT NULL COMMENT '文章标题（最长200字符）',
  heat INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '热度（浏览量，无符号防负数）',
  like_count INT UNSIGNED NOT NULL DEFAULT 0 COMMENT '点赞数（无符号防负数）',
  master_tag VARCHAR(100) COMMENT '主标签（最长100字符）',
  sub_tag VARCHAR(100) COMMENT '副标签（最长100字符）',
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间（自动记录插入时间）',
  PRIMARY KEY (id),
  -- 外键约束：关联自身表 id，父表删除时子表 father_id 置为 NULL
  FOREIGN KEY (father_id) REFERENCES blog_content_main(id) ON DELETE SET NULL ON UPDATE CASCADE,
  INDEX idx_create_time (create_time),
  INDEX idx_author_id (author_id),
  INDEX idx_father_id (father_id),
  INDEX idx_master_tag (master_tag),
  UNIQUE INDEX uk_biz_id (biz_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='博客详细内容表';
